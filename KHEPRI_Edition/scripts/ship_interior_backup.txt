class_name ShipInterior
extends Control

# =============================================================================
# Walk-the-ship interior ("Explore mode").
# You control an astronaut with the arrow keys / WASD, walk around the deck and
# stand on a station pad. Press E (or Space) to "check" that station: you get a
# real-life science explanation and the engine room lets you power it up/down.
# This is the friendly, hands-on way to inspect the ship before you fly off.
# =============================================================================

signal room_inspected(sys: String)
signal power_requested(sys: String, delta: int)

# 5 columns x 2 rows of station rooms, laid out like the ship diagram.
const STATIONS := [
	"bio", "oxygen", "shields", "weapons", "piloting",
	"engines", "repair", "sensors", "comms", "medbay",
]
const ROW_0 := ["bio", "oxygen", "shields", "weapons", "piloting"]
const ROW_1 := ["engines", "repair", "sensors", "comms", "medbay"]

const PLAYER_SPEED := 260.0
const INTERACT_COOLDOWN := 0.35

var gs: GameState
var checked: Dictionary = {}          # sys -> true once visited this session
var input_gate: Callable = Callable() # returns true when input should be ignored
var help_text := "Walk with the arrow keys or WASD. Stand on a station and press E to check it."

var _player: Sprite2D
var _crew_nodes: Array = []
var _pos := Vector2.ZERO
var _t := 0.0
var _near := ""
var _cooldown := 0.0
var _deck_cache := Rect2()
var _prompt_card: PanelContainer
var _prompt_label: Label
var _progress_label: Label


func _ready() -> void:
	custom_minimum_size = Vector2(560, 360)
	mouse_filter = Control.MOUSE_FILTER_PASS
	texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST

	_player = Sprite2D.new()
	_player.texture = load("res://assets/generated/crew_astronaut.png")
	_player.scale = Vector2(0.7, 0.7)
	_player.texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
	_player.z_index = 10
	add_child(_player)

	_build_footer()
	resized.connect(_relayout_crew)
	set_process(true)


func _build_footer() -> void:
	_prompt_card = PanelContainer.new()
	_prompt_card.add_theme_stylebox_override("panel", UiTheme.card(UiTheme.GOLD, 0.25, 20))
	_prompt_card.visible = false
	_prompt_card.mouse_filter = Control.MOUSE_FILTER_IGNORE
	add_child(_prompt_card)
	_prompt_label = Label.new()
	_prompt_label.add_theme_font_size_override("font_size", 20)
	_prompt_label.add_theme_color_override("font_color", UiTheme.TEXT)
	_prompt_card.add_child(_prompt_label)

	_progress_label = Label.new()
	_progress_label.add_theme_font_size_override("font_size", 18)
	_progress_label.add_theme_color_override("font_color", UiTheme.GOLD)
	_progress_label.mouse_filter = Control.MOUSE_FILTER_IGNORE
	add_child(_progress_label)


func set_state(state: GameState) -> void:
	gs = state
	checked.clear()
	_setup_crew()
	_relayout_crew()
	_refresh_progress()


func _setup_crew() -> void:
	for n in _crew_nodes:
		n.queue_free()
	_crew_nodes.clear()
	if gs == null:
		return
	var cols := [Color(0.7, 0.9, 1.0), Color(1.0, 0.85, 0.6), Color(0.8, 1.0, 0.8), Color(1.0, 0.8, 0.9)]
	for i in gs.crew.size():
		var s := Sprite2D.new()
		s.texture = load("res://assets/generated/crew_astronaut.png")
		s.scale = Vector2(0.42, 0.42)
		s.texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
		s.modulate = cols[i % cols.size()]
		s.z_index = 8
		add_child(s)
		_crew_nodes.append(s)


# --- Layout -----------------------------------------------------------------
func _deck_rect() -> Rect2:
	var w := maxf(size.x, 420.0)
	var h := maxf(size.y, 300.0)
	return Rect2(24, 54, w - 48, h - 132)


func _station_center(sys: String) -> Vector2:
	var d := _deck().position
	var sz := _deck().size
	var cw := sz.x / 5.0
	var ch := sz.y / 2.0
	var idx := STATIONS.find(sys)
	if idx < 0:
		return d + sz * 0.5
	var col := idx % 5
	var row := floori(float(idx) / 5.0)
	return d + Vector2((col + 0.5) * cw, (row + 0.5) * ch)


func _deck() -> Rect2:
	if _deck_cache == Rect2():
		_deck_cache = _deck_rect()
	return _deck_cache


func _card_size() -> Vector2:
	var d := _deck()
	return Vector2(clampf(d.size.x / 5.0 - 26.0, 90.0, 168.0), clampf(d.size.y / 2.0 - 30.0, 80.0, 132.0))


func _relayout_crew() -> void:
	_deck_cache = _deck_rect()
	if _pos == Vector2.ZERO:
		_pos = _deck().position + _deck().size * Vector2(0.5, 0.86)
	if _player:
		_player.position = _pos
	for i in _crew_nodes.size():
		if _crew_nodes[i].position == Vector2.ZERO:
			_crew_nodes[i].position = _deck().position + _deck().size * Vector2(randf(), randf())


# --- Input / update ---------------------------------------------------------
func _process(delta: float) -> void:
	_t += delta
	_deck_cache = _deck_rect()
	if _cooldown > 0.0:
		_cooldown -= delta
	var blocked: bool = input_gate.is_valid() and bool(input_gate.call())
	if not blocked and gs != null:
		var dir := Vector2.ZERO
		if Input.is_action_pressed("move_left"):
			dir.x -= 1.0
		if Input.is_action_pressed("move_right"):
			dir.x += 1.0
		if Input.is_action_pressed("move_up"):
			dir.y -= 1.0
		if Input.is_action_pressed("move_down"):
			dir.y += 1.0
		if dir != Vector2.ZERO:
			dir = dir.normalized()
			_pos += dir * PLAYER_SPEED * delta
			var pad := 20.0
			_pos.x = clampf(_pos.x, _deck().position.x + pad, _deck().end.x - pad)
			_pos.y = clampf(_pos.y, _deck().position.y + pad, _deck().end.y - pad)
		if Input.is_action_just_pressed("interact") and _cooldown <= 0.0:
			_try_interact()
	_update_near()
	_update_nodes()
	queue_redraw()


func _update_nodes() -> void:
	if _player:
		_player.position = _pos
		var bob := 1.0 if UiTheme.reduce_motion else (0.7 + 0.04 * sin(_t * 4.0))
		_player.scale = Vector2(bob, bob)
	for i in _crew_nodes.size():
		if _crew_nodes[i].position == Vector2.ZERO:
			_crew_nodes[i].position = _deck().position + _deck().size * Vector2(randf(), randf())


func _station_at(p: Vector2) -> String:
	var cs := _card_size() * 0.5 + Vector2(14, 14)
	for sys in STATIONS:
		var c := _station_center(sys)
		var r := Rect2(c - cs, cs * 2.0)
		if r.has_point(p):
			return sys
	return ""


func _update_near() -> void:
	var sys := _station_at(_pos)
	_near = sys
	if _prompt_card == null:
		return
	if sys == "":
		_prompt_card.visible = false
		return
	_prompt_card.visible = true
	_prompt_label.text = "Press  E  to check:  %s" % _pretty(sys)
	_prompt_card.reset_size()
	_prompt_card.position = Vector2(
		clampf(_station_center(sys).x - _prompt_card.size.x * 0.5, 8.0, size.x - _prompt_card.size.x - 8.0),
		_deck().end.y + 6.0)


func _try_interact() -> void:
	if _near == "":
		return
	_cooldown = INTERACT_COOLDOWN
	checked[_near] = true
	_refresh_progress()
	room_inspected.emit(_near)


func request_power(sys: String, delta: int) -> void:
	power_requested.emit(sys, delta)


func _refresh_progress() -> void:
	if _progress_label == null:
		return
	_progress_label.text = "Stations checked:  %d / %d" % [checked.size(), STATIONS.size()]
	_progress_label.position = Vector2(24, 18)


func _pretty(sys: String) -> String:
	return str(ShipInterior.PRETTY.get(sys, sys.capitalize()))


const PRETTY := {
	"piloting": "Cockpit", "engines": "Engine Room", "oxygen": "Oxygen Control",
	"shields": "Shield Generator", "weapons": "Tractor Beam", "sensors": "Science Lab",
	"comms": "Comms Room", "bio": "BioNutrients Bay", "repair": "Repair Bay", "medbay": "Medical Bay",
}


# --- Draw -------------------------------------------------------------------
func _draw() -> void:
	var d := _deck()
	# deck floor with a soft glow outline
	draw_rect(d.grow(6.0), Color(0.10, 0.16, 0.40, 0.55), true)
	draw_rect(d.grow(6.0), UiTheme.CYAN, false, 3.0)
	# floor tiles
	var tile := 44.0
	var gx := 0.0
	while gx < d.size.x:
		draw_line(d.position + Vector2(gx, 0), d.position + Vector2(gx, d.size.y), Color(1, 1, 1, 0.05), 1.0)
		gx += tile
	var gy := 0.0
	while gy < d.size.y:
		draw_line(d.position + Vector2(0, gy), d.position + Vector2(d.size.x, gy), Color(1, 1, 1, 0.05), 1.0)
		gy += tile
	var f := ThemeDB.fallback_font
	var cs := _card_size()
	for sys in STATIONS:
		var c := _station_center(sys)
		var rect := Rect2(c - cs * 0.5, cs)
		var col := _sys_color(sys)
		var is_near: bool = sys == _near
		var done: bool = checked.has(sys)
		# rounded-ish station pad (a filled rect plus an inner border)
		draw_rect(rect, Color(col.r, col.g, col.b, 0.30 if is_near else 0.16), true)
		draw_rect(rect, col, false, 4.0 if is_near else 2.0)
		if is_near:
			draw_rect(rect.grow(4.0), Color(UiTheme.GOLD.r, UiTheme.GOLD.g, UiTheme.GOLD.b, 0.85), false, 3.0)
		# glyph in a bright disc
		var glyph := _glyph(sys)
		draw_circle(Vector2(c.x, rect.position.y + 32.0), 20.0, Color(col.r, col.g, col.b, 0.9))
		var gsz := f.get_string_size(glyph, HORIZONTAL_ALIGNMENT_LEFT, -1, 22)
		draw_string(f, Vector2(c.x - gsz.x * 0.5, rect.position.y + 40.0), glyph, HORIZONTAL_ALIGNMENT_LEFT, -1, 22, Color(0.06, 0.08, 0.2))
		# label
		var nm := _pretty(sys)
		var words := nm.split(" ", false)
		var l1 := nm
		var l2 := ""
		if words.size() > 1:
			var mid := int(ceil(float(words.size()) / 2.0))
			l1 = " ".join(words.slice(0, mid))
			l2 = " ".join(words.slice(mid))
		_draw_centered(f, l1, Vector2(c.x, rect.position.y + 66.0), 14, UiTheme.TEXT)
		if l2 != "":
			_draw_centered(f, l2, Vector2(c.x, rect.position.y + 82.0), 14, UiTheme.TEXT)
		# checkmark when visited
		if done:
			draw_circle(Vector2(rect.end.x - 14.0, rect.position.y + 14.0), 11.0, UiTheme.GOOD)
			_draw_centered(f, "v", Vector2(rect.end.x - 14.0, rect.position.y + 19.0), 16, Color(0.05, 0.2, 0.1))
	# little name tags over the crew
	for i in _crew_nodes.size():
		var p: Vector2 = _crew_nodes[i].position
		if gs != null and i < gs.crew.size():
			var nm := str(gs.crew[i]["name"])
			_draw_centered(f, nm, p + Vector2(0, -20), 12, UiTheme.TEXT_DIM)


func _draw_centered(f: Font, text: String, center: Vector2, fs: int, col: Color) -> void:
	var sz := f.get_string_size(text, HORIZONTAL_ALIGNMENT_LEFT, -1, fs)
	draw_string(f, center + Vector2(-sz.x * 0.5, 0), text, HORIZONTAL_ALIGNMENT_LEFT, -1, fs, col)


func _glyph(sys: String) -> String:
	for s in GameData.SYSTEMS:
		if s["id"] == sys:
			return str(s["icon"])
	return "?"


func _sys_color(sys: String) -> Color:
	if gs == null:
		return UiTheme.CYAN
	var integ := float(gs.sys_integrity.get(sys, 1.0))
	if integ < 0.35:
		return Color(1.0, 0.35, 0.35)
	if not gs.sys_online(sys):
		return Color(0.6, 0.62, 0.72)
	if integ < 0.75:
		return Color(1.0, 0.8, 0.3)
	return Color(0.3, 0.9, 1.0)
